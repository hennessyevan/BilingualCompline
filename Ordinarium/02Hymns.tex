\firstlinetranslation{\GreColored{1.} Before the fading of light, we pray you, O Creator of the universe, with your accustomed mercy to be [our] protector and guardian.}

\GreColored{2.} Te corda nostra sómnient,\newline
te per sopórem séntiant,\newline
tuámque semper glóriam\newline
vicína luce cóncinant. &
\GreColored{3.} Vitam salúbrem tríbue,\newline
nostrum calórem réfice\newline
tætram noctis calíginem\newline
tua collústret cláritas. &
\GreColored{4.} Præsta, Pater omnípotens,\newline
per Iesum Christum Dóminum,\newline
qui tecum in perpétuum\newline
regnat cum Sancto Spíritu.\newline
\hspace*{\saveparindent}Amen. & \\

\small\textit{\GreColored{2.} May our hearts dream of you,\newline
May they perceive you in sleep.\newline
And, with the approach of light,\newline
May they always celebrate your glory.} &
\small\textit{\GreColored{3.} Grant us a wholesome life,\newline
Restore our ardour.\newline
May your brilliance light up\newline
The over-powering darkness\newline
\hspace*{\saveparindent}of the night.} &
\small\textit{\GreColored{4.} Grant this, Almighty Father,\newline
Through the Lord Jesus Christ,\newline
Who reigns with you for ever\newline
With the Holy Spirit. \newline
\hspace*{\saveparindent}Amen.} & \\
\end{tabularx}

\firstlinetranslation{\GreColored{1.} O Christ, brightness and day, who draw back the darkness of night, and are deemed to be light of light, announcing light to the blessed,}

\noindent\begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X@{}}
\GreColored{2.} Precámur, sancte Dómine,\newline
hac nocte nos custódias;\newline
sit nobis in te réquies,\newline
quiétas horas tríbue. &
\GreColored{3.} Somno si dantur óculi,\newline
cor semper ad te vígilet;\newline
tuáque dextra prótegas\newline
fidéles, qui te díligunt. &
\GreColored{4.} Defénsor noster, áspice,\newline
insidiántes réprime,\newline
gubérna tuos fámulos,\newline
quos sánguine mercátus es. &
\GreColored{5.} Sit, Christe, rex piíssime,\newline
tibi Patríque glória,\newline
cum Spíritu Paráclito,\newline
in sempitérna sæcula.\newline
\hspace*{\saveparindent}Amen. \\

\small\textit{\GreColored{2.} We pray, O holy Lord,\newline
That you guard us this night;\newline
May our rest be in you.\newline
Grant us tranquil hours.} &
\small\textit{\GreColored{3.} Were our eyes to slumber,\newline
May our heart be always awake to you.\newline
And may your right hand protect\newline
The faithful who love you.} &
\small\textit{\GreColored{4.} O defender of ours, take note:\newline
Curb those who lay snares.\newline
Guide your servants\newline
Whom you purchased\newline
\hspace*{\saveparindent}with [your] Blood.} &
\small\textit{\GreColored{5.} Glory to you, O Christ,\newline
\hspace*{\saveparindent}most merciful King,\newline
And to the Father,\newline
With the Spirit Paraclete\newline
For ever and ever. Amen.} \\
\end{tabularx}

\subsubsection{Iesu redemptor sæculi}

\chantlabel{Tempus Paschale:}\hfill\chantlabelenglish{Easter Time:}
\gregorioscore{../Hymns/15084-hy--iesu_redemptor_saeculi_tp--solesmes_1983-no-signs.gabc}

\chantlabel{Dominica Pentecostes:}\hfill\chantlabelenglish{Pentecost Sunday:}
\gregorioscore{../Hymns/4676-hy--iesu_redemptor_(dominica_pentecostes)--vatican-no-signs.gabc}

\firstlinetranslation{\GreColored{1.} O Jesus, Redeemer of the world, Word of the Father most high, light of light invisible, ever watchful guardian of your people:}

\noindent\begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X@{}}
\GreColored{2.} Tu fabricátor ómnium\newline
discrétor atque témporum,\newline
fessa labóre córpora\newline
noctis quiéte récrea. &
\GreColored{3.} Qui frangis ima tártara,\newline
tu nos ab hoste líbera,\newline
ne váleat sedúcere\newline
tuo redémptos sánguine, &
\GreColored{4.} Ut, dum graváti córpore\newline
brevi manémus témpore,\newline
sic caro nostra dórmiat\newline
ut mens sopórem nésciat. &
\GreColored{5.} Iesu, tibi sit glória,\newline
qui morte victa prænites,\newline
cum Patre et almo Spíritu,\newline
in sempitérna sæcula. Amen. \\

\small\textit{\GreColored{2.} You, the Maker of all things\newline
And Judge of [all] ages,\newline
Restore by the rest of night\newline
{[Our]} bodies worn out with toil.} &
\small\textit{\GreColored{3.} You, who shatter the depths\newline
\hspace*{\saveparindent}of the underworld,\newline
Free us from the enemy\newline
Lest he be able to deceive\newline
Those redeemed by your blood,} &
\small\textit{\GreColored{4.} So that, while for a short time\newline
We remain weighed down by our body,\newline
Our flesh may thus fall asleep,\newline
So that our soul may know no slumber.} &
\small\textit{\GreColored{5.} O Jesus, glory to you,\newline
Who, when death is conquered,\newline
\hspace*{\saveparindent}shine brightly,\newline
With the Father\newline
\hspace*{\saveparindent}and the gracious Spirit,\newline
For ever and evermore. Amen.} \\
\end{tabularx}
